Python
import streamlit as st
import datetime

st.set_page_config(page_title="LegalTech: Generador de NDA", layout="centered")

st.title("⚖️ Generador Automático de NDA Bilingüe")
st.write("Ingresa los datos del cliente para generar el acuerdo de confidencialidad al instante.")

# Formulario de entrada
with st.form("nda_form"):
    cliente = st.text_input("Nombre de la Firma Legal (Ej. Galicia Abogados)")
    fecha = st.date_input("Fecha del Acuerdo", datetime.date.today())
    submit_button = st.form_submit_button(label="Generar Documento")

# Generación del documento
if submit_button and cliente:
    # Plantilla HTML inyectando los datos
    html_content = f"""
    <html>
    <head><meta charset="utf-8"></head>
    <body style="font-family: Georgia, serif; line-height: 1.6; color: #333;">
        <h2 style="text-align: center;">MUTUAL NON-DISCLOSURE AGREEMENT<br>ACUERDO MUTUO DE CONFIDENCIALIDAD</h2>
        <p><strong>Effective Date / Fecha de Vigencia:</strong> {fecha.strftime('%d-%m-%Y')}</p>
        <hr>
        <p>This Agreement is entered into by <strong>Patrick Bennett</strong> ("Consultant") and <strong>{cliente}</strong> ("Client").</p>
        <p><em>Este Acuerdo es celebrado entre <strong>Patrick Bennett</strong> ("Consultor") y <strong>{cliente}</strong> ("Cliente").</em></p>
        <p><strong>1. Purpose / Propósito:</strong> The Parties are exploring a business relationship regarding legal data auditing.<br>
        <em>Las Partes exploran una relación comercial sobre auditoría de datos legales.</em></p>
        <br><br>
        <table style="width: 100%; text-align: center; margin-top: 50px;">
            <tr>
                <td>_______________________<br><strong>Patrick Bennett</strong><br>Consultor</td>
                <td>_______________________<br><strong>{cliente}</strong><br>Cliente</td>
            </tr>
        </table>
    </body>
    </html>
    """
    
    st.success(f"¡Documento para {cliente} generado con éxito!")
    
    # Botón para descargar
    st.download_button(
        label="📥 Descargar NDA (HTML para imprimir como PDF)",
        data=html_content,
        file_name=f"NDA_Patrick_Bennett_{cliente.replace(' ', '_')}.html",
        mime="text/html"
    )
